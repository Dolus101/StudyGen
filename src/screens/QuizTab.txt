import { useState } from 'react';
import { Pressable, StyleSheet, Text, View } from 'react-native';
import { Ionicons, MaterialCommunityIcons } from '@expo/vector-icons';
// import { FLASHCARDS } from '../data/sampleReviewers';

const C = { card: '#12152E', line: '#1C2044', text: '#FFFFFF', muted: '#8E92B2', accent: '#6B6CFF', button: '#5B63F0' };

export default function FlashcardsTab({ r }) {
  const base = FLASHCARDS[r.id] || [];
  const [order, setOrder] = useState(base.map((_, i) => i));
  const [pos, setPos] = useState(0);
  const [flipped, setFlipped] = useState(false);

  if (!base.length) return <Text style={s.muted}>No flashcards yet</Text>;

  const card = base[order[pos]];
  const topic = r.topics[card.chapter] || r.topics[0];
  const TIcon = topic.lib === 'mci' ? MaterialCommunityIcons : Ionicons;

  const go = (d) => {
    setPos((p) => (p + d + order.length) % order.length);
    setFlipped(false);
  };
  const shuffle = () => {
    const a = [...order];
    for (let i = a.length - 1; i > 0; i--) {
      const j = Math.floor(Math.random() * (i + 1));
      [a[i], a[j]] = [a[j], a[i]];
    }
    setOrder(a);
    setPos(0);
    setFlipped(false);
  };

  return (
    <View>
      <View style={s.head}>
        <View>
          <Text style={s.h}>Flashcards</Text>
          <Text style={s.muted}>{r.stats.flashcards} cards · All chapters</Text>
        </View>
        <Pressable onPress={shuffle} style={({ pressed }) => [s.shuffle, pressed && { opacity: 0.8 }]}>
          <Ionicons name="shuffle" size={14} color={C.accent} />
          <Text style={s.shuffleText}>Shuffle</Text>
        </Pressable>
      </View>

      <Pressable style={s.chapter}>
        <View style={s.badge}>
          <Text style={s.badgeText}>{String(card.chapter + 1).padStart(2, '0')}</Text>
        </View>
        <View style={{ flex: 1 }}>
          <Text style={s.chapterLabel}>CHAPTER {card.chapter + 1}</Text>
          <Text style={s.chapterTitle} numberOfLines={1}>{topic.title}</Text>
        </View>
        <Ionicons name="chevron-down" size={18} color={C.muted} />
      </Pressable>

      <View style={s.countRow}>
        <Text style={s.count}>Card {pos + 1} of {r.stats.flashcards}</Text>
        <Text style={s.muted}>{flipped ? 'Answer side' : 'Question side'}</Text>
      </View>
      <View style={s.track}>
        <View style={[s.fill, { width: `${((pos + 1) / order.length) * 100}%` }]} />
      </View>

      <Pressable onPress={() => setFlipped((f) => !f)} accessibilityLabel="Flip card" style={s.flash}>
        <View style={s.flashTop}>
          <Text style={s.concept}>{flipped ? 'ANSWER' : 'KEY CONCEPT'}</Text>
          <Ionicons name="bookmark-outline" size={18} color={C.muted} />
        </View>
        <View style={s.flashBody}>
          {!flipped && (
            <View style={s.flashIcon}>
              <TIcon name={topic.icon} size={26} color={C.accent} />
            </View>
          )}
          <Text style={s.flashText}>{flipped ? card.a : card.q}</Text>
          {!flipped && <Text style={[s.muted, { textAlign: 'center', marginTop: 8 }]}>Think of your answer before you reveal it.</Text>}
        </View>
        <View style={s.flip}>
          <Ionicons name="sync-outline" size={13} color={C.muted} />
          <Text style={s.flipText}>Tap the card to flip</Text>
        </View>
      </Pressable>

      <Pressable onPress={() => setFlipped((f) => !f)} style={({ pressed }) => [s.primary, pressed && { opacity: 0.85 }]}>
        <Ionicons name="eye-outline" size={18} color="#fff" />
        <Text style={s.primaryText}>{flipped ? 'Show question' : 'Reveal answer'}</Text>
      </Pressable>

      <View style={s.nav}>
        <Pressable onPress={() => go(-1)} style={({ pressed }) => [s.navBtn, pressed && { opacity: 0.8 }]}>
          <Ionicons name="chevron-back" size={16} color={C.text} />
          <Text style={s.navText}>Previous</Text>
        </Pressable>
        <Pressable onPress={() => go(1)} style={({ pressed }) => [s.navBtn, pressed && { opacity: 0.8 }]}>
          <Text style={s.navText}>Next card</Text>
          <Ionicons name="chevron-forward" size={16} color={C.text} />
        </Pressable>
      </View>

      <View style={s.tip}>
        <Text style={s.tipTitle}>Practice active recall</Text>
        <Text style={s.muted}>After revealing the answer, mark what you know or keep the card for another review.</Text>
      </View>
    </View>
  );
}

const s = StyleSheet.create({
  muted: { color: C.muted, fontSize: 12 },
  head: { flexDirection: 'row', justifyContent: 'space-between', alignItems: 'center' },
  h: { color: C.text, fontSize: 16, fontWeight: '700', marginBottom: 2 },
  shuffle: { flexDirection: 'row', alignItems: 'center', gap: 6, paddingHorizontal: 12, height: 32, borderRadius: 9, backgroundColor: '#1A1E4A' },
  shuffleText: { color: C.accent, fontSize: 12, fontWeight: '600' },

  chapter: { flexDirection: 'row', alignItems: 'center', gap: 12, marginTop: 14, padding: 12, borderRadius: 14, backgroundColor: '#0F1230', borderWidth: 1, borderColor: '#262B63' },
  badge: { width: 38, height: 38, borderRadius: 10, backgroundColor: C.button, alignItems: 'center', justifyContent: 'center' },
  badgeText: { color: '#fff', fontWeight: '700', fontSize: 13 },
  chapterLabel: { color: C.accent, fontSize: 10, fontWeight: '700', letterSpacing: 0.6 },
  chapterTitle: { color: C.text, fontSize: 14, fontWeight: '600', marginTop: 2 },

  countRow: { flexDirection: 'row', justifyContent: 'space-between', marginTop: 16 },
  count: { color: C.text, fontSize: 13, fontWeight: '700' },
  track: { height: 4, borderRadius: 2, backgroundColor: '#23274D', marginTop: 8, overflow: 'hidden' },
  fill: { height: 4, borderRadius: 2, backgroundColor: C.accent },

  flash: { marginTop: 14, minHeight: 250, borderRadius: 18, padding: 16, backgroundColor: '#0F1230', borderWidth: 1, borderColor: '#262B63', justifyContent: 'space-between' },
  flashTop: { flexDirection: 'row', justifyContent: 'space-between', alignItems: 'center' },
  concept: { color: C.accent, fontSize: 11, fontWeight: '700', letterSpacing: 0.8 },
  flashBody: { alignItems: 'center', paddingHorizontal: 8 },
  flashIcon: { width: 54, height: 54, borderRadius: 14, backgroundColor: '#1B1F4A', alignItems: 'center', justifyContent: 'center', marginBottom: 14 },
  flashText: { color: C.text, fontSize: 19, fontWeight: '600', textAlign: 'center', lineHeight: 26 },
  flip: { flexDirection: 'row', alignItems: 'center', justifyContent: 'center', gap: 6 },
  flipText: { color: C.muted, fontSize: 11 },

  primary: { flexDirection: 'row', gap: 8, height: 48, borderRadius: 12, backgroundColor: C.button, alignItems: 'center', justifyContent: 'center', marginTop: 14 },
  primaryText: { color: '#fff', fontSize: 15, fontWeight: '700' },
  nav: { flexDirection: 'row', gap: 10, marginTop: 10 },
  navBtn: { flex: 1, flexDirection: 'row', gap: 6, height: 46, borderRadius: 12, backgroundColor: C.card, borderWidth: 1, borderColor: C.line, alignItems: 'center', justifyContent: 'center' },
  navText: { color: C.text, fontSize: 13, fontWeight: '500' },

  tip: { marginTop: 14, padding: 14, borderRadius: 14, backgroundColor: C.card, gap: 4 },
  tipTitle: { color: C.text, fontSize: 13, fontWeight: '700' },
});
